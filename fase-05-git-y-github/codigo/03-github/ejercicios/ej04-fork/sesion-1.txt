mkdir maria && cd maria && git init
git config user.name "María Soto"
git config user.email maria@example.com
printf '# Recetario\n\nGazpacho: tomate, pepino, ajo, aceite y vinagr.\n' > README.md
git add README.md

git push -u origin main
cd ..
